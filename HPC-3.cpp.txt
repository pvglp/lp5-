#include <iostream>
#include <omp.h>
#include <vector>
#include <climits>
using namespace std;

//-----MINIMUM-----
void minimum(vector<int>& array)
{
    int min_val = INT_MAX;

    double start = omp_get_wtime();

    for (int i=0;i<array.size();i++)
    {
        if (array[i] < min_val)
	{
	    min_val = array[i];
	}
    }

    double end = omp_get_wtime();

    cout<<"\nMinimum (Sequential) : "<<min_val<<endl;
    cout<<"Time : "<<end-start<<endl;

    int min_ele = INT_MAX;
    start = omp_get_wtime();

    #pragma omp parallel for reduction(min:min_ele)
    for (int i=0;i<array.size();i++)
    {
        if (array[i] < min_ele)
	{
	    min_ele = array[i];
	}
    }
    end = omp_get_wtime();
    
    cout<<"\nMinimum (Parallel) : "<<min_val<<endl;
    cout<<"Time : "<<end-start<<endl;
}

//-----MAXIMUM-----
void maximum(vector<int>& array)
{
    int max_val = INT_MIN;

    double start = omp_get_wtime();

    for (int i=0;i<array.size();i++)
    {
        if (array[i] > max_val)
	{
	    max_val = array[i];
	}
    }

    double end = omp_get_wtime();

    cout<<"\nMaximum (Sequential) : "<<max_val<<endl;
    cout<<"Time : "<<(end-start)<<endl;

    int max_ele = INT_MIN;
    start = omp_get_wtime();

    #pragma omp parallel for reduction(max:max_ele)
    for (int i=0;i<array.size();i++)
    {
        if (array[i] > max_ele)
	{
	    max_ele = array[i];
	}
    }
    end = omp_get_wtime();
    
    cout<<"\nMaximum (Parallel) : "<<max_val<<endl;
    cout<<"Time : "<<(end-start)<<endl;
}

//----SUM----
void sum(vector<int>& array)
{
    long long total = 0;
    double start = omp_get_wtime();
    for(int i=0;i<array.size();i++)
    {
	total = total + array[i];
    }
    double end = omp_get_wtime();
    cout<<"\nSum (Sequential) : "<<total<<endl;
    cout<<"Time : "<<(end-start)<<endl;

    total = 0;
    start = omp_get_wtime();
    #pragma omp parallel for reduction(+:total)
    for (int i=0;i<array.size();i++)
    {
	total = total + array[i];
    }
    end = omp_get_wtime();
    cout<<"\nSum (Parallel) : "<<total<<endl;
    cout<<"Time : "<<(end-start)<<endl;
}

//----AVERAGE----
void average(vector<int>& array)
{
    long long total = 0;
    double start = omp_get_wtime();
    for(int i=0;i<array.size();i++)
    {
	total = total + array[i];
    }
    double end = omp_get_wtime();
    cout<<"\nAverage (Sequential) : "<<(double)total/array.size()<<endl;
    cout<<"Time : "<<(end-start)<<endl;

    total = 0;
    start = omp_get_wtime();
    #pragma omp parallel for reduction(+:total)
    for (int i=0;i<array.size();i++)
    {
	total = total + array[i];
    }
    end = omp_get_wtime();
    cout<<"\nAverage (Parallel) : "<<(double)total/array.size()<<endl;
    cout<<"Time : "<<(end-start)<<endl;
}

int main()
{
    int N;
    int MAX = 1000;
    cout<<"Enter number of elements : ";
    cin>>N;

    vector<int> array(N);

    for (int i=0;i<N;i++)
    {
	array[i] = rand() % MAX;
	//cin>>array[i];
    }

    minimum(array);
    maximum(array);
    sum(array);
    average(array);

    return 0;
}
